### 3.5 创建 `ai_service.py`

python

```
import re
import json
import jieba
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

def load_skills(path="skill_dict.txt"):
    with open(path, "r", encoding="utf-8") as f:
        return set(line.strip() for line in f if line.strip())

def extract_skills(text, skill_set):
    found = set()
    for skill in skill_set:
        if skill.lower() in text.lower():
            found.add(skill)
    return sorted(found)

def extract_education(text):
    if "博士" in text:
        return "博士"
    if "硕士" in text or "研究生" in text:
        return "硕士"
    if "本科" in text or "学士" in text:
        return "本科"
    if "专科" in text or "大专" in text:
        return "专科"
    return "不限"

def extract_years(text):
    patterns = [r"(\d+)\s*年经验", r"工作\s*(\d+)\s*年", r"(\d+)\s*年"]
    for p in patterns:
        m = re.search(p, text)
        if m:
            return int(m.group(1))
    return 0

def jieba_tokenize(text):
    return list(jieba.cut(text))

def text_similarity(text1, text2):
    vectorizer = TfidfVectorizer(tokenizer=jieba_tokenize, token_pattern=None)
    tfidf = vectorizer.fit_transform([text1, text2])
    sim = cosine_similarity(tfidf[0:1], tfidf[1:2])[0][0]
    return float(sim)

def calc_edu_score(resume_edu, job_edu):
    order = {"不限": 0, "专科": 1, "本科": 2, "硕士": 3, "博士": 4}
    r = order.get(resume_edu, 0)
    j = order.get(job_edu, 0)
    if j == 0 or r >= j:
        return 1.0
    return 0.5

def calc_years_score(resume_years, job_years):
    if job_years <= 0:
        return 1.0
    diff = job_years - resume_years
    if diff <= 0:
        return 1.0
    if diff == 1:
        return 0.7
    if diff == 2:
        return 0.4
    return 0.2

def build_suggestion(missing_skills, edu_score, years_score, text_score):
    parts = []
    if missing_skills:
        parts.append("建议补充 " + "、".join(missing_skills[:3]) + " 项目经验。")
    if edu_score < 1:
        parts.append("建议突出相关课程、项目或继续教育经历。")
    if years_score < 1:
        parts.append("建议强化项目深度和实习经历。")
    if text_score < 0.3:
        parts.append("建议在简历中使用岗位 JD 中的关键词。")
    return "".join(parts) or "简历与岗位匹配良好，可进一步突出项目成果。"

def match_resume_job(resume, job):
    resume_skills = set(json.loads(resume.skills))
    job_skills = set(json.loads(job.skills))

    matched = sorted(resume_skills & job_skills)
    missing = sorted(job_skills - resume_skills)
    skill_score = len(matched) / len(job_skills) if job_skills else 1.0

    text_score = text_similarity(resume.text, job.text)
    edu_score = calc_edu_score(resume.education, job.education)
    years_score = calc_years_score(resume.years, job.years)

    score = round((skill_score * 0.4 + text_score * 0.3 + edu_score * 0.2 + years_score * 0.1) * 100, 1)

    if score >= 80:
        level = "高度匹配"
    elif score >= 65:
        level = "较匹配"
    elif score >= 50:
        level = "一般"
    else:
        level = "不匹配"

    suggestion = build_suggestion(missing, edu_score, years_score, text_score)

    return {
        "score": score,
        "level": level,
        "skill_score": round(skill_score, 4),
        "text_score": round(text_score, 4),
        "edu_score": edu_score,
        "years_score": years_score,
        "matched_skills": matched,
        "missing_skills": missing,
        "suggestion": suggestion
    }
```



### 3.6 创建 `main.py`

python

```
import json
from fastapi import FastAPI, Depends, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from sqlalchemy.orm import Session

import models
import schemas
from database import engine, get_db, SessionLocal
from ai_service import load_skills, extract_skills, extract_education, extract_years, match_resume_job

models.Base.metadata.create_all(bind=engine)

app = FastAPI(title="智能简历岗位匹配系统")

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

SKILLS = load_skills()

@app.get("/api/health")
def health():
    return {"status": "ok"}

@app.post("/api/resumes")
def create_resume(data: schemas.ResumeCreate, db: Session = Depends(get_db)):
    skills = extract_skills(data.text, SKILLS)
    education = extract_education(data.text)
    years = extract_years(data.text)
    resume = models.Resume(
        name=data.name,
        text=data.text,
        skills=json.dumps(skills, ensure_ascii=False),
        education=education,
        years=years
    )
    db.add(resume)
    db.commit()
    db.refresh(resume)
    return {
        "id": resume.id,
        "name": resume.name,
        "skills": skills,
        "education": education,
        "years": years
    }

@app.get("/api/resumes")
def list_resumes(page: int = 1, size: int = 10, db: Session = Depends(get_db)):
    query = db.query(models.Resume).order_by(models.Resume.id.desc())
    total = query.count()
    items = query.offset((page - 1) * size).limit(size).all()
    return {
        "total": total,
        "items": [
            {
                "id": r.id,
                "name": r.name,
                "skills": json.loads(r.skills),
                "education": r.education,
                "years": r.years,
                "created_at": r.created_at
            } for r in items
        ]
    }

@app.post("/api/jobs")
def create_job(data: schemas.JobCreate, db: Session = Depends(get_db)):
    skills = extract_skills(data.text, SKILLS)
    education = extract_education(data.text)
    years = extract_years(data.text)
    job = models.Job(
        title=data.title,
        company=data.company,
        text=data.text,
        skills=json.dumps(skills, ensure_ascii=False),
        education=education,
        years=years
    )
    db.add(job)
    db.commit()
    db.refresh(job)
    return {
        "id": job.id,
        "title": job.title,
        "company": job.company,
        "skills": skills,
        "education": education,
        "years": years
    }

@app.get("/api/jobs")
def list_jobs(page: int = 1, size: int = 10, db: Session = Depends(get_db)):
    query = db.query(models.Job).order_by(models.Job.id.desc())
    total = query.count()
    items = query.offset((page - 1) * size).limit(size).all()
    return {
        "total": total,
        "items": [
            {
                "id": j.id,
                "title": j.title,
                "company": j.company,
                "skills": json.loads(j.skills),
                "education": j.education,
                "years": j.years,
                "created_at": j.created_at
            } for j in items
        ]
    }

@app.post("/api/match")
def match(data: schemas.MatchRequest, db: Session = Depends(get_db)):
    resume = db.query(models.Resume).filter(models.Resume.id == data.resume_id).first()
    job = db.query(models.Job).filter(models.Job.id == data.job_id).first()
    if not resume or not job:
        raise HTTPException(status_code=404, detail="简历或岗位不存在")

    result = match_resume_job(resume, job)

    record = models.MatchRecord(
        resume_id=resume.id,
        job_id=job.id,
        score=result["score"],
        level=result["level"],
        skill_score=result["skill_score"],
        text_score=result["text_score"],
        edu_score=result["edu_score"],
        years_score=result["years_score"],
        matched_skills=json.dumps(result["matched_skills"], ensure_ascii=False),
        missing_skills=json.dumps(result["missing_skills"], ensure_ascii=False),
        suggestion=result["suggestion"]
    )
    db.add(record)
    db.commit()
    db.refresh(record)

    result["id"] = record.id
    result["created_at"] = record.created_at
    return result

@app.get("/api/records")
def list_records(page: int = 1, size: int = 10, db: Session = Depends(get_db)):
    query = db.query(models.MatchRecord).order_by(models.MatchRecord.id.desc())
    total = query.count()
    items = query.offset((page - 1) * size).limit(size).all()
    return {
        "total": total,
        "items": [
            {
                "id": r.id,
                "resume_id": r.resume_id,
                "job_id": r.job_id,
                "score": r.score,
                "level": r.level,
                "created_at": r.created_at
            } for r in items
        ]
    }

@app.get("/api/stats")
def stats(db: Session = Depends(get_db)):
    records = db.query(models.MatchRecord).all()
    total = len(records)
    avg_score = round(sum(r.score for r in records) / total, 1) if total else 0
    level_distribution = {"高度匹配": 0, "较匹配": 0, "一般": 0, "不匹配": 0}
    missing_counter = {}

    for r in records:
        level_distribution[r.level] = level_distribution.get(r.level, 0) + 1
        for skill in json.loads(r.missing_skills):
            missing_counter[skill] = missing_counter.get(skill, 0) + 1

    top5 = sorted(missing_counter.items(), key=lambda x: x[1], reverse=True)[:5]
    return {
        "total": total,
        "avg_score": avg_score,
        "level_distribution": level_distribution,
        "missing_skill_top5": [{"skill": k, "count": v} for k, v in top5]
    }
```
